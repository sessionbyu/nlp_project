```
from sqlalchemy.orm import Session
from app.models.prediction import Prediction

def save_prediction(db: Session, text: str, label: str, score: float):
    record = Prediction(text=text, label=label, score=score)
    db.add(record)
    db.commit()
    db.refresh(record)
    return record

def get_history(db: Session, skip: int = 0, limit: int = 20):
    return db.query(Prediction).order_by(Prediction.created_at.desc()).offset(skip).limit(limit).all()
```

### 3. 认证服务 (`services/auth_service.py`)

python

```
from datetime import datetime, timedelta
from jose import JWTError, jwt
from passlib.context import CryptContext
from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from sqlalchemy.orm import Session
from app.core.database import get_db
from app.models.user import User
from app.core.config import settings

SECRET_KEY = "your-secret-key-change-in-prod"
ALGORITHM = "HS256"
ACCESS_TOKEN_EXPIRE_MINUTES = 30

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")
oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/api/v1/auth/login")

def verify_password(plain, hashed):
    return pwd_context.verify(plain, hashed)

def get_password_hash(password):
    return pwd_context.hash(password)

def create_access_token(data: dict):
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
    to_encode.update({"exp": expire})
    return jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)

def get_current_user(token: str = Depends(oauth2_scheme), db: Session = Depends(get_db)):
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
    )
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        email: str = payload.get("sub")
        if email is None:
            raise credentials_exception
    except JWTError:
        raise credentials_exception
    user = db.query(User).filter(User.email == email).first()
    if user is None:
        raise credentials_exception
    return user
```





### 4. 认证路由 (`api/v1/auth.py`)

python

```
from fastapi import APIRouter, Depends, HTTPException, status
from fastapi.security import OAuth2PasswordRequestForm
from sqlalchemy.orm import Session
from app.core.database import get_db
from app.services.auth_service import (
    get_password_hash, verify_password, create_access_token, get_current_user
)
from app.models.user import User

router = APIRouter(prefix="/auth", tags=["Authentication"])

@router.post("/register")
def register(email: str, password: str, db: Session = Depends(get_db)):
    if db.query(User).filter(User.email == email).first():
        raise HTTPException(status_code=400, detail="Email already registered")
    user = User(email=email, hashed_password=get_password_hash(password))
    db.add(user)
    db.commit()
    return {"msg": "User created"}

@router.post("/login")
def login(form_data: OAuth2PasswordRequestForm = Depends(), db: Session = Depends(get_db)):
    user = db.query(User).filter(User.email == form_data.username).first()
    if not user or not verify_password(form_data.password, user.hashed_password):
        raise HTTPException(status_code=400, detail="Incorrect email or password")
    token = create_access_token({"sub": user.email})
    return {"access_token": token, "token_type": "bearer"}
```

## 详细设计

### 1. 角色权限管理（RBAC）

**目标**：管理员可管理所有用户数据，普通用户仅能操作自己的记录。
**位置**：`models/user.py` 增加 `role` 字段；`services/auth_service.py` 增加权限检查；`api/v1/admin.py` 新建管理员路由。
**关键代码**：

python

```
# models/user.py
class User(Base):
    role = Column(String(10), default="user")  # "admin" / "user"

# services/auth_service.py
def require_role(required: str):
    def checker(current_user: User = Depends(get_current_user)):
        if current_user.role != required:
            raise HTTPException(403, "Permission denied")
        return current_user
    return checker

# api/v1/admin.py
@router.get("/all-history")
def get_all_history(admin=Depends(require_role("admin")), db=Depends(get_db)):
    return db.query(Prediction).all()
```

### 2. API Key 认证

**目标**：为第三方开发者提供 API Key 调用方式，替代 JWT。
**方案**：增加 `ApiKey` 模型，在 `core/auth.py` 中增加 `api_key_header` 依赖，检查请求头 `X-API-Key`。
**关键**：生成随机 key（`secrets.token_urlsafe`），关联用户，支持停用/启用。

### 3. 邮箱验证与密码重置

**目标**：注册验证邮箱，忘记密码时发送重置链接。
**方案**：使用 `aiosmtplib` 发送邮件，生成限时 token（`itsdangerous`）。路由：`/verify-email`, `/forgot-password`, `/reset-password`。
**模型**：`EmailVerification`, `PasswordResetToken`。

### 4. 高级搜索与过滤

**目标**：按标签、日期范围、文本关键词过滤历史。
**位置**：`api/v1/history.py`，`services/history_service.py`。
**关键代码**：

python

```
# services/history_service.py
def search(db, user_id, label=None, start_date=None, end_date=None, keyword=None, skip=0, limit=20):
    q = db.query(Prediction).filter(Prediction.user_id == user_id)
    if label: q = q.filter(Prediction.label == label)
    if start_date: q = q.filter(Prediction.created_at >= start_date)
    if end_date: q = q.filter(Prediction.created_at <= end_date)
    if keyword: q = q.filter(Prediction.text.ilike(f"%{keyword}%"))
    return q.order_by(Prediction.created_at.desc()).offset(skip).limit(limit).all()
```





### 5. 结果导出

**目标**：将搜索结果导出为 CSV/JSON。
**路由**：`GET /api/v1/history/export?format=csv`，返回 `StreamingResponse`。
**工具**：`csv` 模块或 `pandas`。

### 6. 软删除与归档

**目标**：用户删除记录时不物理删除，标记为 `is_deleted=True`。
**模型**：`Prediction` 增加 `is_deleted`、`deleted_at`。查询时默认过滤已删除。可提供回收站恢复功能。

### 7. 文件上传批量分析

**目标**：上传 CSV/Excel，自动解析文本列，返回批量结果。
**实现**：`POST /api/v1/upload`，接收文件，使用 `openpyxl` 或 `pandas` 解析，调用 `predict_batch`，保存结果并返回下载链接。
**文件处理**：存入临时目录或对象存储（MinIO）。

### 8. 大文件分块上传

**目标**：支持 >100MB 文件的可靠上传。
**方案**：前端使用 `tus` 协议分块，后端 `tuspy` 库处理，接口 `/api/v1/upload/chunk`。

### 9. Celery 异步推理

**目标**：批量任务或大文件分析不阻塞请求。
**实现**：引入 `celery` + Redis/RabbitMQ，任务提交后返回 `task_id`，客户端轮询结果。
**服务层**：`services/celery_tasks.py` 定义 `analyze_large_file` 任务。

### 10. WebSocket 实时进度通知

**目标**：异步任务执行时，前端收到进度百分比。
**方案**：`fastapi.WebSocket` 端点，Celery 任务通过 Redis pub/sub 推送进度，WebSocket 转发给客户端。

### 11. Redis 缓存

**目标**：相同文本重复请求时直接返回缓存。
**修改**：`services/inference.py` 的 `predict_safe` 中，先查 Redis，未命中再推理并缓存（key = MD5(text)）。使用 `redis-py`。

### 12. API 限流

**目标**：限制每用户/IP 的请求频率。
**方案**：使用 `slowapi`（基于 `ratelimit`），装饰器 `@limiter.limit("5/minute")` 作用于路由。

### 13. 请求/响应日志

**目标**：记录所有 API 调用详情，便于审计。
**实现**：自定义 FastAPI 中间件 `middleware/logging.py`，记录用户、时间、路径、请求体、响应状态、耗时到日志文件或数据库表 `api_logs`。

## 整合建议

- **按优先级实施**：先做用户相关的完善（角色、邮箱、API Key），再加强数据功能（搜索、导出、软删除），其次性能（缓存、限流、异步），最后是监控和前端。
- **保持分层**：每个新功能遵循 `models → services → api/v1 → main.py` 的顺序，避免跨层依赖。
- **数据库迁移**：使用 `Alembic` 管理模型变更，替代之前的 `Base.metadata.create_all`。
- **环境变量**：所有敏感配置（密钥、邮件账号、Redis 地址）通过环境变量注入，不硬编码。

通过以上20个功能点，你的 NLP 项目将升级为一个**生产级、可商用的 Web 服务平台**，具备完善的业务闭环。每个功能均可独立开发、测试、上线，并可与现有代码无缝集成。
